actualizar ubuntu:	sudo apt update 
aplicar ubuntu:		sudo apt upgrade

Instalar MYSQL: sudo apt install mysql-server
Verificar versión: mysql --versión
Lanzar mysql server siempre que se levante la instacion de EC2: sudo systemctl enable mysql

MySQL:
sudo mysql
CREATE DATABASE IF NOT EXISTS db_canopus;
CREATE USER 'canopus'@'localhost' IDENTIFIED BY 'DEFINE_UNA_CONTRASENA_SEGURA';
GRANT SELECT, INSERT, UPDATE, DELETE, CREATE, ALTER, INDEX, DROP, REFERENCES ON db_canopus.* TO 'canopus'@'localhost';
flush privileges;
exit
cd /etc/mysql/mysql.conf.d
sudo nano mysqld.cnf
